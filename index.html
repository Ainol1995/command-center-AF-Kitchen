/** * ============================================================================ * CLOUD COMMAND CENTER — GOOGLE APPS SCRIPT BACKEND (Code.gs) * ============================================================================ */ // 1. TUKAR KOD PIN DI BAWAH KEPADA PIN 4-6 DIGIT RAHSIA ANDA SENDIRI: const SECRET_PIN = "123456"; // 2. Tetapan Peringatan Google Calendar (dalam minit) const REMINDER_BAKING_MINS = 24 * 60; // 24 jam sebelum delivery const REMINDER_PACKING_MINS = 2 * 60; // 2 jam sebelum delivery function doGet(e) { try { const params = (e && e.parameter) ? e.parameter : {}; const pin = String(params.pin || "").trim(); if (pin !== SECRET_PIN) { return jsonResponse({ ok: false, error: "PIN_INVALID", message: "Kod PIN salah. Akses disekat." }); } const ss = SpreadsheetApp.getActiveSpreadsheet(); ensureSheetsExist(ss); return jsonResponse({ ok: true, serverTime: new Date().toISOString(), data: readFullState(ss) }); } catch (err) { return jsonResponse({ ok: false, error: "SERVER_ERROR", message: err.toString() }); } } function doPost(e) { try { if (!e || !e.postData || !e.postData.contents) { return jsonResponse({ ok: false, error: "EMPTY_BODY", message: "Tiada data dihantar." }); } const req = JSON.parse(e.postData.contents); const pin = String(req.pin || "").trim(); if (pin !== SECRET_PIN) { return jsonResponse({ ok: false, error: "PIN_INVALID", message: "Kod PIN salah. Tindakan ditolak." }); } const ss = SpreadsheetApp.getActiveSpreadsheet(); ensureSheetsExist(ss); const action = req.action; if (action === "saveConfig") { saveConfigToSheets(ss, req.config); return jsonResponse({ ok: true, message: "Tetapan bisnes & senarai bahan berjaya disimpan ke Cloud.", data: readFullState(ss) }); } if (action === "createOrder") { const result = createOrderInCloud(ss, req.order, req.updatedIngredients); return jsonResponse({ ok: true, message: "Tempahan direkod, stok ditolak & peringatan Google Calendar dijadualkan.", calendarCreated: result.calendarCreated, data: readFullState(ss) }); } if (action === "markOrderPaid") { updateOrderPayment(ss, req.orderId); return jsonResponse({ ok: true, message: "Status bayaran dikemas kini kepada Lunas.", data: readFullState(ss) }); } if (action === "markOrderDone") { updateOrderCompleted(ss, req.orderId); return jsonResponse({ ok: true, message: "Tempahan ditandakan selesai.", data: readFullState(ss) }); } if (action === "updateStock") { syncIngredientsSheet(ss, req.ingredients); return jsonResponse({ ok: true, message: "Baki stok bahan dikemas kini.", data: readFullState(ss) }); } return jsonResponse({ ok: false, error: "UNKNOWN_ACTION", message: "Tindakan tidak dikenali." }); } catch (err) { return jsonResponse({ ok: false, error: "SERVER_ERROR", message: err.toString() }); } } function ensureSheetsExist(ss) { let ordersSheet = ss.getSheetByName("Orders"); if (!ordersSheet) { ordersSheet = ss.insertSheet("Orders"); const headers = [ "Order_ID", "Pelanggan", "Tarikh_Delivery", "Masa", "Saiz_ID", "Saiz_Nama", "Perisa", "Add_On", "Jumlah_Harga", "Bayaran_Diterima", "Status_Bayaran", "Status_Kerja", "Calendar_Event_ID", "Dikemaskini_Pada" ]; ordersSheet.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight("bold").setBackground("#fef3c7"); ordersSheet.setFrozenRows(1); } let ingSheet = ss.getSheetByName("Ingredients"); if (!ingSheet) { ingSheet = ss.insertSheet("Ingredients"); const headers = [ "Ing_ID", "Nama_Bahan", "Kategori", "Unit", "Berat_Pek", "Harga_Pek", "Stok_Semasa", "Amaran_Stok_Rendah" ]; ingSheet.getRange(1, 1, 1, headers.length).setValues([headers]).setFontWeight("bold").setBackground("#fef3c7"); ingSheet.setFrozenRows(1); } let cfgSheet = ss.getSheetByName("Config_JSON"); if (!cfgSheet) { cfgSheet = ss.insertSheet("Config_JSON"); cfgSheet.getRange(1, 1, 1, 2).setValues([["Config_Data_JSON", "Last_Updated"]]).setFontWeight("bold").setBackground("#f5f5f4"); cfgSheet.setFrozenRows(1); } } function readFullState(ss) { const cfgSheet = ss.getSheetByName("Config_JSON"); const rawCfg = cfgSheet.getRange("A2").getValue(); let configObj = null; if (rawCfg) { try { configObj = JSON.parse(rawCfg); } catch (e) { configObj = null; } } const ingSheet = ss.getSheetByName("Ingredients"); const ingLastRow = ingSheet.getLastRow(); const ingredients = []; if (ingLastRow >= 2) { const rows = ingSheet.getRange(2, 1, ingLastRow - 1, 8).getValues(); rows.forEach(r => { if (r[0]) { ingredients.push({ id: String(r[0]), name: String(r[1]), category: String(r[2] || "Bahan Asas"), unit: String(r[3] || "g"), packQty: Number(r[4] || 1), packPrice: Number(r[5] || 0), stock: Number(r[6] || 0), lowAlert: Number(r[7] || 0) }); } }); } if (configObj && ingredients.length > 0) { configObj.ingredients = ingredients; } const ordersSheet = ss.getSheetByName("Orders"); const ordLastRow = ordersSheet.getLastRow(); const orders = []; if (ordLastRow >= 2) { const rows = ordersSheet.getRange(2, 1, ordLastRow - 1, 14).getValues(); rows.forEach(r => { if (r[0]) { orders.push({ id: String(r[0]), customer: String(r[1]), date: formatSheetDate(r[2]), time: formatSheetTime(r[3]), sizeId: String(r[4]), sizeName: String(r[5]), flavours: String(r[6] || "").split(",").map(s => s.trim()).filter(Boolean), addons: String(r[7] || "").split(",").map(s => s.trim()).filter(Boolean), total: Number(r[8] || 0), paidAmount: Number(r[9] || 0), paymentStatus: String(r[10] || "Paid"), status: String(r[11] || "Active"), calendarEventId: String(r[12] || "") }); } }); } return { config: configObj, orders: orders }; } function saveConfigToSheets(ss, cfg) { if (!cfg) return; const cfgToStore = Object.assign({}, cfg); delete cfgToStore.pin; const cfgSheet = ss.getSheetByName("Config_JSON"); cfgSheet.getRange("A2:B2").setValues([[JSON.stringify(cfgToStore), new Date().toISOString()]]); if (Array.isArray(cfg.ingredients)) { syncIngredientsSheet(ss, cfg.ingredients); } } function syncIngredientsSheet(ss, ingredients) { const ingSheet = ss.getSheetByName("Ingredients"); const lastRow = ingSheet.getLastRow(); if (lastRow >= 2) { ingSheet.getRange(2, 1, lastRow - 1, 8).clearContent(); } if (!ingredients || ingredients.length === 0) return; const rows = ingredients.map(ing => [ ing.id, ing.name, ing.category || "Bahan Asas", ing.unit || "g", Number(ing.packQty || 1), Number(ing.packPrice || 0), Number(ing.stock || 0), Number(ing.lowAlert || 0) ]); ingSheet.getRange(2, 1, rows.length, 8).setValues(rows); } function createOrderInCloud(ss, ord, updatedIngredients) { let calendarEventId = ""; let calendarCreated = false; try { const dateParts = String(ord.date || "").split("-"); const timeParts = String(ord.time || "12:00").split(":"); if (dateParts.length === 3) { const startTime = new Date( Number(dateParts[0]), Number(dateParts[1]) - 1, Number(dateParts[2]), Number(timeParts[0] || 12), Number(timeParts[1] || 0) ); const endTime = new Date(startTime.getTime() + 60 * 60 * 1000); const flavoursText = (ord.flavours || []).join(" + "); const addonsText = (ord.addons && ord.addons.length) ? ord.addons.join(", ") : "Tiada"; const balance = Number(ord.total || 0) - Number(ord.paidAmount || 0); const payText = balance > 0 ? `Deposit (Baki belum bayar: RM ${balance.toFixed(2)})` : `Lunas (RM ${Number(ord.total || 0).toFixed(2)})`; const title = `🎂 ${ord.sizeName} - ${ord.customer} (${flavoursText})`; const desc = [ `ID Tempahan: ${ord.id}`, `Pelanggan: ${ord.customer}`, `Produk: ${ord.sizeName}`, `Perisa: ${flavoursText}`, `Add-On: ${addonsText}`, `Status Bayaran: ${payText}` ].join("\n"); const cal = CalendarApp.getDefaultCalendar(); const event = cal.createEvent(title, startTime, endTime, { description: desc }); event.removeAllReminders(); event.addPopupReminder(REMINDER_BAKING_MINS); event.addPopupReminder(REMINDER_PACKING_MINS); calendarEventId = event.getId(); calendarCreated = true; } } catch (calErr) { calendarCreated = false; } const ordersSheet = ss.getSheetByName("Orders"); ordersSheet.appendRow([ ord.id, ord.customer, ord.date, ord.time, ord.sizeId, ord.sizeName, (ord.flavours || []).join(", "), (ord.addons || []).join(", "), Number(ord.total || 0), Number(ord.paidAmount || 0), ord.paymentStatus || "Paid", ord.status || "Active", calendarEventId, new Date().toISOString() ]); if (Array.isArray(updatedIngredients)) { syncIngredientsSheet(ss, updatedIngredients); } return { calendarCreated: calendarCreated }; } function updateOrderPayment(ss, orderId) { const ordersSheet = ss.getSheetByName("Orders"); const lastRow = ordersSheet.getLastRow(); if (lastRow < 2) return; const ids = ordersSheet.getRange(2, 1, lastRow - 1, 1).getValues(); for (let i = 0; i < ids.length; i++) { if (String(ids[i][0]) === String(orderId)) { const rowNum = i + 2; const totalVal = Number(ordersSheet.getRange(rowNum, 9).getValue() || 0); ordersSheet.getRange(rowNum, 10).setValue(totalVal); ordersSheet.getRange(rowNum, 11).setValue("Paid"); ordersSheet.getRange(rowNum, 14).setValue(new Date().toISOString()); break; } } } function updateOrderCompleted(ss, orderId) { const ordersSheet = ss.getSheetByName("Orders"); const lastRow = ordersSheet.getLastRow(); if (lastRow < 2) return; const rows = ordersSheet.getRange(2, 1, lastRow - 1, 14).getValues(); for (let i = 0; i < rows.length; i++) { if (String(rows[i][0]) === String(orderId)) { const rowNum = i + 2; ordersSheet.getRange(rowNum, 12).setValue("Completed"); ordersSheet.getRange(rowNum, 14).setValue(new Date().toISOString()); const eventId = String(rows[i][12] || ""); if (eventId) { try { const event = CalendarApp.getDefaultCalendar().getEventById(eventId); if (event) { const oldTitle = event.getTitle(); if (!oldTitle.startsWith("✅")) { event.setTitle("✅ [SELESAI] " + oldTitle); } event.removeAllReminders(); } } catch (e) {} } break; } } } function formatSheetDate(val) { if (val instanceof Date) { return Utilities.formatDate(val, Session.getScriptTimeZone(), "yyyy-MM-dd"); } return String(val || ""); } function formatSheetTime(val) { if (val instanceof Date) { return Utilities.formatDate(val, Session.getScriptTimeZone(), "HH:mm"); } return String(val || "12:00"); } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
